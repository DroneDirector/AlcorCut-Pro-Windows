cmake_minimum_required(VERSION 3.20)
project(alcorcut_mf CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(JNI REQUIRED)
add_library(alcorcut_mf SHARED mfcam.cpp)
target_include_directories(alcorcut_mf PRIVATE ${JNI_INCLUDE_DIRS})
target_link_libraries(alcorcut_mf PRIVATE mfplat mf mfreadwrite mfuuid strmiids wmcodecdspuuid PortableDeviceGUIDs ole32 oleaut32 shlwapi)
target_compile_definitions(alcorcut_mf PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(alcorcut_mf PRIVATE /W3 /O2 /EHsc)
  set_property(TARGET alcorcut_mf PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
endif()
